#+TITLE: Emacs Installation Instructions
#+PROPERTY: header-args :tangle install.sh :shebang #!/usr/bin/env bash

* Installation instructions

Currently, I use version 28 of Emacs. Since it is not officially
released yet, I have to build it myself.

To do so, you need to install quite some third-party packages

#+begin_src shell
  sudo apt install libxpm-dev libgif-dev libjpeg-dev libpng-dev \
       libtiff-dev libx11-dev libncurses5-dev automake autoconf \
       texinfo libgtk2.0-dev libgtk-3-dev gcc-10 g++-10 libgccjit0 \
       libgccjit-10-dev libgccjit-9-dev libjansson4 libjansson-dev \
       libgnutls28-dev libacl1-dev libmagickwand-dev
#+end_src

Then, download and extract the latest Emacs version

#+begin_src shell
  __EMACS_VERSION="28.2"
  __OPT="${HOME}/.local/opt"
  __EMACS_HOME="${__OPT}/emacs-${__EMACS_VERSION}"
  mkdir -p "${__EMACS_HOME}"

  if [ ! -f "${__EMACS_HOME}/bin/emacs" ]; then
  (
      cd "/tmp" || exit 1
      __EMACS_TAR_FILENAME="emacs-${__EMACS_VERSION}.tar.gz"
      wget "http://ftp.fau.de/gnu/emacs/${__EMACS_TAR_FILENAME}"
      tar -xvf "${__EMACS_TAR_FILENAME}" -C "${__EMACS_HOME}"
  )
  fi
#+end_src

To build Emacs, run

#+begin_src shell
    export CC=/usr/bin/gcc-10 CXX=/usr/bin/gcc-10
    (
        cd "${__EMACS_HOME}" || exit
        ./autogen.sh
        ./configure --with-modules --with-gnutls --with-rsvg \
                    --with-x --with-json \
                    --prefix "${__EMACS_HOME}" --bindir "${__EMACS_HOME}/bin" \
                    CFLAGS="-O3 -mtune=native -march=native -fomit-frame-pointer" \
                    --with-mailutils --with-native-compilation --with-imagemagick \
                    --with-jpeg --with-png --with-rsvg --with-tiff --with-wide-int \
                    --with-xft --with-xml2 --with-xpm
        make -j"$(nproc)"
        make install
    )
#+end_src

Finally, make the new version globally available in your system using
update-alternatives.

#+begin_src shell
  sudo update-alternatives --install "/usr/bin/emacsclient" emacsclient "${__EMACS_HOME}/bin/emacsclient" 100
  sudo update-alternatives --install "/usr/bin/emacs" emacs "${__EMACS_HOME}/bin/emacs" 100
  sudo update-alternatives --auto emacs
  sudo update-alternatives --auto emacsclient
#+end_src

Once you have Emacs installed, it configures it on its own at
startup.
