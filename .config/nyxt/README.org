* Install via Docker

#+begin_src bash :results output :shebang #!/usr/bin/env bash
  docker build \
         --build-arg DOCKER_USER=$(whoami) \
         --build-arg UID=$(id -u $(whoami)) \
         --build-arg GID=$(id -g $(whoami)) \
         -t nyxt .
#+end_src

#+begin_src bash :results output :shebang #!/usr/bin/env bash
  docker run --user $(whoami) -it nyxt
#+end_src
