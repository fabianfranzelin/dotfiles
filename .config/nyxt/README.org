* Install via Docker

#+begin_src bash :tangle build.sh :shebang #!/usr/bin/env bash
  export DOTFILES
  DOTFILES=$(git rev-parse --show-toplevel)

  docker build \
         --build-arg USERNAME="$(whoami)" \
         --build-arg UID="$(id -u "$(whoami)")" \
         -f "Dockerfile" \
         -t nyxt \
         "${DOTFILES}"
#+end_src


* Run it

#+BEGIN_SRC bash :tangle run.sh :shebang #!/usr/bin/env bash
docker run --rm \
       --user "$(whoami)" \
       --privileged \
       --net host \
       -e http_proxy \
       -e https_proxy \
       -e DISPLAY \
       -e XAUTHORITY="/tmp/.Xauthority" \
       -e DEBIAN_FRONTEND="noninteractive" \
       -e WEBKIT_DISABLE_COMPOSITING_MODE=1 \
       -e GDK_SYNCHRONIZE=1 \
       -e NO_AT_BRIDGE=1 \
       -v "${HOME}/.Xauthority:/tmp/.Xauthority" \
       -v "/tmp/.X11-unix:/tmp/.X11-unix" \
       -v "/var/run/docker.sock:/var/run/docker.sock" \
       -v "${HOME}/.config/nyxt:${HOME}/.config/nyxt" \
       nyxt:latest \
       "$@"
#+END_SRC

Make the docker command available in the =PATH= and then use
=update-alternatives= to have it in the =dmenu=

#+BEGIN_SRC bash :shebang #!/usr/bin/env bash
sudo update-alternatives --install "/usr/bin/nyxt" nyxt "/usr/local/bin/nyxt" 100
sudo update-alternatives --install "/usr/bin/nyxt" nyxt "${HOME}/.local/bin/nyxt-docker" 200
#+END_SRC
